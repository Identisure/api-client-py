def verify_callback_signature(body: bytes | str, signature: str, secret: str) -> bool: 
10def verify_callback_signature(body: bytes | str, signature: str, secret: str) -> bool:
11    """Verify the ``X-Signature-SHA256`` header of a callback request.
12
13    Pass the raw request body exactly as received. Verify before parsing and
14    never re-serialize the JSON.
15    """
16    if isinstance(body, str):
17        body = body.encode("utf-8")
18    try:
19        expected = base64.b64decode(signature, validate=True)
20    except (binascii.Error, ValueError):
21        return False
22    actual = hmac.new(secret.encode("utf-8"), body, hashlib.sha256).digest()
23    return hmac.compare_digest(actual, expected)
